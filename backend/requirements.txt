# --- API HTTP -------------------------------------------------------------------------------
fastapi
uvicorn[standard]
# Requis par FastAPI dès qu'une route utilise Form/UploadFile. Absent du requirements.txt de la
# v1 alors que le code en dépendait : crash au démarrage, comblé à la main dans le venv puis
# jamais reporté. Il reste ici même si aucune route ne l'utilise encore — le coût est nul, la
# panne qu'il évite ne l'est pas.
python-multipart

# --- Structures et journalisation -----------------------------------------------------------
# pydantic v2 imposé : le planificateur expose ses plans en modèles typés (v1 incompatible).
pydantic>=2
# Paquet SÉPARÉ de pydantic depuis la v2 — `backend/core/config.py` en dépend pour `BaseSettings`.
# Oublié au premier jet, exactement comme python-multipart l'avait été dans la v1 : le conteneur
# démarre, uvicorn meurt à l'import, et l'échec ne se voit que dans les journaux.
pydantic-settings
loguru

# --- Accès externes -------------------------------------------------------------------------
httpx                 # client HTTP asynchrone (recherche Hugging Face, sondes moteurs)
tokenizers            # décompte de contexte pour Strata, qui n'expose pas de /tokenize
# Pour le BAC À SABLE, pas pour le backend. Mesuré le 2026-08-26 : à « upload par API », le modèle
# écrit `import requests` — c'est le réflexe de tout code Python HTTP publié depuis quinze ans — et
# reçoit `ModuleNotFoundError`. Il ne peut pas l'installer lui-même : le bac abaisse ses privilèges
# et pip n'a pas droit d'écriture sur le venv. Il a donc conclu, à tort mais logiquement, que
# « le sandbox ne dispose d'aucun client HTTP qui fonctionne » et a rendu la main.
# httpx était pourtant présent. Le corriger par la consigne serait combattre un réflexe ; l'ajouter
# coûte 200 Kio et supprime le problème.
requests              # client HTTP du bac à sable : ce que le modèle importe spontanément
huggingface_hub       # recherche, métadonnées et téléchargement des modèles
python-dotenv         # lecture du .env en exécution native (hors Docker)

# --- Mesure matérielle (domaine system) -----------------------------------------------------
# Le planificateur mesure, il ne suppose pas : RAM libre réelle et processus via psutil, VRAM
# libre réelle via NVML. NVML est le chemin programmatique officiel — préférable au parsing du
# texte de `nvidia-smi`, dont le format n'est pas un contrat.
psutil
nvidia-ml-py

llama-cpp-python
